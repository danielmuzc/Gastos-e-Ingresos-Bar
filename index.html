Caja de la barra

Sin conectar
Publicar la app para que funcione en cualquier equipo

Una sola vez, desde Google Sheets

  1. Abre tu hoja de cálculo y ve a Extensiones → Apps Script. En el archivo Código.gs pega el script de abajo.
  2. Pulsa + → HTML, nómbralo index y pega allí todo el contenido de este archivo HTML.
  3. Pulsa Implementar → Nueva implementación → Aplicación web. Ejecutar como: Yo. Quién tiene acceso: Cualquier persona.
  4. Autoriza los permisos y abre la URL que termina en /exec: esa es la aplicación, ya conectada. Compártela por WhatsApp o agrégala a la pantalla de inicio del celular.
  5. Si cambias el script o el HTML, crea una nueva versión de la implementación.

El script crea solo las pestañas Ingresos y Gastos con sus encabezados.

const HOJAS = { Ingreso: 'Ingresos', Gasto: 'Gastos' };
const COLUMNAS = ['ID','Fecha','Categoría','Descripción','Método de pago','Monto (COP)'];

function hoja_(tipo) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let s = ss.getSheetByName(HOJAS[tipo]);
  if (!s) {
    s = ss.insertSheet(HOJAS[tipo]);
    s.appendRow(COLUMNAS);
    s.setFrozenRows(1);
    s.getRange('B:B').setNumberFormat('@');
    s.getRange('F:F').setNumberFormat('$ #,##0');
  }
  return s;
}

function salida_(o) {
  return ContentService.createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}

// Abrir la URL /exec muestra la aplicación. Con ?api=1 devuelve los datos en JSON.
function doGet(e) {
  if (e && e.parameter && e.parameter.api === '1') return salida_(leer_());
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('Caja de la barra')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// Para usar el archivo HTML fuera de Apps Script (con la URL /exec en URL_API).
function doPost(e) {
  return salida_(escribir_(JSON.parse(e.postData.contents)));
}

// Funciones que la aplicación llama directamente desde Apps Script.
function leerMovimientos() { return leer_(); }
function guardarMovimiento(b) { return escribir_(b); }

function leer_() {
  const tz = Session.getScriptTimeZone();
  const data = [];
  Object.keys(HOJAS).forEach(tipo => {
    hoja_(tipo).getDataRange().getValues().slice(1).forEach(r => {
      if (r[0] === '') return;
      data.push({
        id: String(r[0]), tipo: tipo,
        fecha: r[1] instanceof Date ? Utilities.formatDate(r[1], tz, 'yyyy-MM-dd') : String(r[1]),
        categoria: r[2], descripcion: r[3], metodo: r[4], monto: Number(r[5])
      });
    });
  });
  return { ok: true, data: data };
}

function escribir_(b) {
  if (!HOJAS[b.tipo]) return { ok: false, error: 'tipo inválido' };
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const s = hoja_(b.tipo);
    const ids = s.getRange(1, 1, s.getLastRow(), 1).getValues().flat().map(String);
    if (b.accion === 'borrar') {
      const i = ids.indexOf(String(b.id));
      if (i > 0) s.deleteRow(i + 1);
    } else {
      // Guardar: si el ID ya existe se actualiza la fila (editar); si no, se agrega.
      const fila = [b.id, b.fecha, b.categoria, b.descripcion, b.metodo, b.monto];
      const i = ids.indexOf(String(b.id));
      if (i > 0) s.getRange(i + 1, 1, 1, 6).setValues([fila]);
      else s.appendRow(fila);
    }
    return { ok: true };
  } finally { lock.releaseLock(); }
}
